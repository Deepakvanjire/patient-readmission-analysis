# ============================================================
# BASIC COUNTS
# ============================================================

Total Patients =
DISTINCTCOUNT(patients[patient_id])


Total Admissions =
COUNTROWS(admissions)


Total Readmissions =
CALCULATE(
    COUNTROWS(admissions),
    admissions[readmitted_30d] = 1
)


Total 7-Day Readmissions =
CALCULATE(
    COUNTROWS(admissions),
    admissions[readmitted_7d] = 1
)


# ============================================================
# RATES
# ============================================================

Readmission Rate =
DIVIDE(
    [Total Readmissions],
    [Total Admissions],
    0
)


Readmission Rate % =
FORMAT([Readmission Rate], "0.00%")


7-Day Readmission Rate =
DIVIDE(
    [Total 7-Day Readmissions],
    [Total Admissions],
    0
)


# ============================================================
# LENGTH OF STAY
# ============================================================

Average LOS =
AVERAGE(admissions[los_days])


Avg LOS Readmitted =
CALCULATE(
    AVERAGE(admissions[los_days]),
    admissions[readmitted_30d] = 1
)


Avg LOS Not Readmitted =
CALCULATE(
    AVERAGE(admissions[los_days]),
    admissions[readmitted_30d] = 0
)


# ============================================================
# FINANCIAL
# ============================================================

Total Cost INR =
SUM(billing[total_cost_inr])


Average Cost Per Admission =
DIVIDE(
    [Total Cost INR],
    [Total Admissions],
    0
)


Total Govt Subsidy =
SUM(billing[govt_subsidy_inr])


Total Out of Pocket =
SUM(billing[out_of_pocket_inr])


Avg Out of Pocket =
AVERAGE(billing[out_of_pocket_inr])


Readmission Total Cost =
CALCULATE(
    SUM(billing[total_cost_inr]),
    admissions[readmitted_30d] = 1
)


Average Cost Readmitted =
CALCULATE(
    AVERAGE(billing[total_cost_inr]),
    admissions[readmitted_30d] = 1
)


Average Cost Not Readmitted =
CALCULATE(
    AVERAGE(billing[total_cost_inr]),
    admissions[readmitted_30d] = 0
)


Readmission Cost Premium =
[Average Cost Readmitted] - [Average Cost Not Readmitted]
